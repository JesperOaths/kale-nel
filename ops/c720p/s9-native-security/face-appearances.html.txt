<script id="s9-faces-appearances-gallery-v1">
window.GEJAST_PAGE_VERSION='v817';
(()=>{
'use strict';
const $=id=>document.getElementById(id);
const clipOK=/^motion_[0-9]{13}[.]mp4$/;
const cropOK=/^motion_[0-9]{13}__[A-Za-z0-9_-]{1,65}__[0-9]{1,9}_[0-2][.]jpg$/;
const types={face:'Face crop',appearance:'Appearance',review:'Person clip'};
const colours={black:'#252d34',white:'#eff4fa',gray:'#94a3b8',red:'#d45757',orange:'#e5964b',yellow:'#ecd05d',green:'#59ae7e',blue:'#5694d2',purple_or_pink:'#ae8ccc',brown:'#97715b'};
const state={data:[],items:[],kind:'all',page:1,query:'',color:'',busy:false,observer:null};
const make=(tag,value,cls)=>{const e=document.createElement(tag);if(value!==undefined)e.textContent=String(value);if(cls)e.className=cls;return e;};
const relay=()=>{
 if(window.C720PSecureRelay?.fetch && window.C720PSecureRelay?.url)return window.C720PSecureRelay;
 try{if(window.parent!==window&&window.parent.location.origin===window.location.origin&&window.parent.C720PSecureRelay?.fetch&&window.parent.C720PSecureRelay?.url)return window.parent.C720PSecureRelay;}catch(_){}
 return null;
};
function safeSnap(rec,snap){return clipOK.test(String(rec.clip_no||'')) && cropOK.test(String(snap||'')) && String(snap).startsWith(rec.clip_no.slice(0,-4)+'__');}
function clipStamp(rec){const name=String(rec.clip_no||'');return clipOK.test(name)?Number(name.slice(7,20)):0;}
function displayTime(rec){
 const ms=clipStamp(rec);
 if(ms){try{return new Intl.DateTimeFormat('en-GB',{timeZone:'Europe/Amsterdam',day:'2-digit',month:'short',year:'numeric',hour:'2-digit',minute:'2-digit'}).format(new Date(ms));}catch(_){}}
 return String(rec.timestamp||'Unknown date');
}
function addChip(host,name,klass){
 const b=make('span',name,'chip'+(klass?' '+klass:''));host.append(b);return b;
}
function read(){
 const items=[];
 for(const rec of state.data){
  const timestamp=clipStamp(rec),tracks=Array.isArray(rec.anonymous_tracks)?rec.anonymous_tracks:[];
  const faces=Array.isArray(rec.face_candidates)?rec.face_candidates:[];
  for(const f of faces.slice(0,8)){
   const snap=safeSnap(rec,f.snapshot_name)?f.snapshot_name:null;
   if(!snap)continue;
   items.push({kind:'face',rec,face:f,snap,timestamp,search:[rec.clip_no,f.person_id,f.candidate_name,f.match_status,rec.timestamp].join(' ').toLowerCase()});
  }
  for(const t of tracks.slice(0,64)){
   const id=String(t.id||'');
   if(!/^Person (?:[1-9]|[1-5][0-9]|6[0-4])$/.test(id))continue;
   const colour=Object.hasOwn(colours,t.upper_clothing_colour)?t.upper_clothing_colour:'';
   items.push({kind:'appearance',rec,track:t,colour,timestamp,search:[rec.clip_no,rec.timestamp,id,colour].join(' ').toLowerCase()});
  }
  // A missing frontal face is a review opportunity, not an identified person.
  if(Number(rec.person_count)>0 && faces.every(f=>!safeSnap(rec,f.snapshot_name))){
   items.push({kind:'review',rec,timestamp,search:[rec.clip_no,rec.timestamp,rec.face_review_status,'needs review'].join(' ').toLowerCase()});
  }
 }
 items.sort((a,b)=>b.timestamp-a.timestamp || (a.kind==='face'?-1:1));
 state.items=items;
 const faceCount=items.filter(x=>x.kind==='face').length;
 const appearanceCount=items.filter(x=>x.kind==='appearance').length;
 const personClips=state.data.filter(x=>Number(x.person_count)>0).length;
 const waiting=state.data.filter(x=>Number(x.person_count)>0 && !(x.face_candidates||[]).some(f=>safeSnap(x,f.snapshot_name))).length;
 $('facesCount').textContent=String(faceCount);
 $('sightingsCount').textContent=String(appearanceCount);
 $('personClipsCount').textContent=String(personClips);
 $('waitingCount').textContent=String(waiting);
 const older=state.data.filter(r=>Number(r.person_count)>0 && clipStamp(r)>0 && clipStamp(r)<1791687597000);
 const deepDone=older.filter(r=>Number(r.face_review_sampled_frames||0)===36).length;
 const percent=older.length?Math.round(100*deepDone/older.length):0;
 $('scanSummary').textContent='Historical face rescue · '+deepDone+' of '+older.length+' eligible recordings rescanned';
 $('scanBar').style.width=percent+'%';
 $('scanMeter').setAttribute('aria-valuenow',String(percent));
 $('scanCaption').textContent='36 time positions per clip · '+(older.length-deepDone)+' remaining · scanned locally by the S9+ when cool and idle. A negative face scan does not prove the video lacks people.';
}
function addVisual(card,item){
 const v=make('div',undefined,'visual');const badge=make('span',types[item.kind],'type');v.append(badge);
 const snap=item.kind==='face'?item.snap:
   item.rec.snapshot_name && /^motion_[0-9]{13}[.]mp4[.]thumb[.]jpg$/.test(String(item.rec.snapshot_name))?item.rec.snapshot_name:null;
 if(snap){
  const img=make('img');img.alt=item.kind==='face'?'Frontal face crop, unverified person candidate':'Video scene thumbnail, NOT an isolated face';
  img.loading='lazy';img.decoding='async';img.dataset.relayPath=item.kind==='face'?'/new/saved/face/'+snap:'/new/saved/snap/'+snap;
  v.append(img);state.observer?.observe(img);
 }else v.append(make('small',item.kind==='face'?'Face image unavailable':'No isolated face crop. View original video for context.'));
 card.append(v);
}
function addTrackClues(card,item){
 const tr=item.track,clips=make('div',undefined,'chips');
 addChip(clips,String(tr.id||'Person (clip only)'));
 if(item.colour){
  const badge=addChip(clips,'Clothing: '+item.colour.replaceAll('_',' '));
  const swatch=make('i',undefined,'swatch');swatch.style.backgroundColor=colours[item.colour];badge.prepend(swatch);
 }
 card.append(clips);
 card.append(make('div','Observed from '+Number(tr.sample_count||0)+' sampled frames · '+(Number(tr.first_sample_ms||0)/1000).toFixed(1)+'–'+(Number(tr.last_sample_ms||0)/1000).toFixed(1)+' s','subtle'));
}
async function play(rec,at,button){
 const r=relay();if(!r||!clipOK.test(String(rec.clip_no||'')))return;
 button.disabled=true;
 try{
  const url=await r.url('/new/saved/clip/'+encodeURIComponent(rec.clip_no));
  const vid=$('player'),dlg=$('playerDialog');
  $('playerLabel').textContent=displayTime(rec)+' · '+rec.clip_no;
  vid.src=url;vid.load();
  const atSeconds=Number(at)/1000;
  if(Number.isFinite(atSeconds)&&atSeconds>0)vid.addEventListener('loadedmetadata',()=>{
   try{vid.currentTime=Math.max(0,Math.min(atSeconds,Math.max(0,vid.duration-0.25)));}catch(_){}
  },{once:true});
  dlg.showModal();
 }catch(_){button.textContent='Recording unavailable';}
 finally{button.disabled=false;}
}
function cardFor(item){
 const rec=item.rec,card=make('article',undefined,'card');addVisual(card,item);
 const body=make('div',undefined,'card-body');
 body.append(make('h3',item.kind==='face'
  ?(typeof item.face.candidate_name==='string'&&item.face.candidate_name?'Candidate: '+item.face.candidate_name:'Anonymous face candidate')
  :item.kind==='appearance'?'Clothing & movement clue':'Person detected · face not captured'));
 body.append(make('div',displayTime(rec),'subtle'));
 if(item.kind==='face'){
  const chips=make('div',undefined,'chips');
  addChip(chips,String(item.face.person_id||'Anonymous candidate'));
  addChip(chips,'Unverified similarity','warn');
  body.append(chips);
  if(item.face.match_status)body.append(make('div',String(item.face.match_status).replaceAll('_',' '),'subtle'));
  if(Number.isFinite(item.face.time_ms))body.append(make('div','Captured at '+(item.face.time_ms/1000).toFixed(1)+' s of clip','subtle'));
 }else if(item.kind==='appearance')addTrackClues(body,item);
 else{
  const chips=make('div',undefined,'chips');
  const deep=Number(rec.face_review_sampled_frames||0)===36;
  addChip(chips,deep?'36-frame face scan complete':'Deep face scan pending',deep?'':'warn');
  body.append(chips);
  body.append(make('div',deep?'36 frame positions checked on the S9+. No suitable frontal face found; the clip may still contain someone.':'No suitable frontal crop in the initial sample. Awaiting a deeper on-phone review.','subtle'));
 }
 body.append(make('div',rec.clip_no,'subtle'));
 const actions=make('div',undefined,'actions'),watch=make('button','Play original clip');
 watch.type='button';watch.addEventListener('click',()=>play(rec,item.kind==='face'?item.face.time_ms:item.kind==='appearance'?item.track.first_sample_ms:0,watch));
 actions.append(watch);body.append(actions);card.append(body);return card;
}
function filtered(){
 return state.items.filter(x=>(state.kind==='all'||state.kind===x.kind) &&
   (!state.color||x.colour===state.color) && (!state.query||x.search.includes(state.query)));
}
function render(){
 state.observer?.disconnect();
 state.observer=('IntersectionObserver' in window)?new IntersectionObserver(entries=>{
  for(const e of entries){if(!e.isIntersecting)continue;state.observer.unobserve(e.target);loadImage(e.target);}
 },{rootMargin:'200px'}):null;
 const matches=filtered(),limit=state.page*24,chosen=matches.slice(0,limit),gallery=$('gallery');
 gallery.replaceChildren();
 if(!chosen.length){
  const note=state.kind==='face'?'No saved frontal-face crops match these filters. S9+ person videos remain available in Appearance and Needs review.':
    'No matching evidence. Try another filter or review newer S9+ motion clips.';
  gallery.append(make('div',note,'empty'));
 }
 for(const item of chosen)gallery.append(cardFor(item));
 $('count').textContent='Showing '+chosen.length+' of '+matches.length+' evidence entries. Entries are not counts of unique people.';
 $('more').hidden=chosen.length>=matches.length;
 if(!state.observer)for(const img of gallery.querySelectorAll('img[data-relay-path]'))loadImage(img);
}
async function loadImage(img){
 if(!img.isConnected||img.dataset.loaded==='1')return;
 img.dataset.loaded='1';
 const path=img.dataset.relayPath;
 if(!path||!/^\/new\/saved\/(face|snap)\/[A-Za-z0-9._-]+$/.test(path))return;
 try{const r=relay();if(!r?.url)throw Error('relay');const signed=await r.url(path);if(img.isConnected)img.src=signed;}
 catch(_){if(img.isConnected){img.remove();}}
}
async function refresh(){
 if(state.busy)return;state.busy=true;
 $('state').textContent='Reading S9+ microSD index…';
 try{
  const r=relay();if(!r?.fetch)throw Error('authenticated_relay_unavailable');
  const response=await r.fetch('/new/api/saved',{cache:'no-store'});
  if(!response.ok)throw Error('archive_unavailable');
  const payload=await response.json();
  if(payload.archive_mode!=='S9-microSD-only'||!Array.isArray(payload.events))throw Error('archive_contract_invalid');
  state.data=payload.events.filter(e=>clipOK.test(String(e.clip_no||'')));
  read();render();$('state').textContent='S9+ archive connected · '+state.data.length+' clips';
 }catch(_){
  $('state').textContent='Camera archive unavailable';
  if(!state.items.length)$('gallery').replaceChildren(make('div','The authenticated archive cannot be reached. Original recording continues independently.','empty'));
 }finally{state.busy=false;}
}
$('filters').addEventListener('click',event=>{
 const b=event.target.closest('button[data-kind]');if(!b)return;
 state.kind=b.dataset.kind;state.page=1;
 for(const n of document.querySelectorAll('button[data-kind]'))n.setAttribute('aria-pressed',String(n===b));
 render();
});
$('search').addEventListener('input',e=>{state.query=String(e.target.value||'').toLowerCase().trim().slice(0,120);state.page=1;render();});
$('colorFilter').addEventListener('change',e=>{state.color=e.target.value;state.page=1;render();});
for(const [id] of Object.entries(colours)){
 const opt=make('option',id.replaceAll('_',' '));opt.value=id;$('colorFilter').append(opt);
}
$('refresh').addEventListener('click',refresh);
$('more').addEventListener('click',()=>{state.page++;render();});
function stopVideo(){const v=$('player');v.pause();v.removeAttribute('src');v.load();}
$('closePlayer').addEventListener('click',()=>$('playerDialog').close());
$('playerDialog').addEventListener('close',stopVideo);
document.addEventListener('visibilitychange',()=>{if(document.hidden)stopVideo();else refresh();});
window.addEventListener('pagehide',stopVideo);
refresh();
window.setInterval(()=>{if(!document.hidden)refresh();},90000);
})();
